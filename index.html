// --- POPRAWIONA INTEGRACJA Z GEMINI API --- els.btnAskAi.addEventListener('click', async () => { const apiKey = els.apiKey.value.trim(); const task = els.aiTask.value.trim(); if (!apiKey) { alert('Wklej swój klucz Gemini API!'); return; } if (!task) { alert('Wpisz treść zadania!'); return; } els.aiOutput.style.display = 'block'; els.aiOutput.innerText = '⏳ Gemini analizuje zadanie...'; try { // Używamy stabilnego i uniwersalnego endpointu dla gemini-1.5-flash (działa z każdym darmowym kluczem z AI Studio) const response = await fetch(`https://generativelanguage.googleapis.com/v1/models/gemini-1.5-flash:generateContent?key=${apiKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: "Rozwiąż to zadanie matematyczne krok po kroku po polsku:\n\n" + task }] }] }) }); const data = await response.json(); if (data.error) { throw new Error(data.error.message || 'Błąd API'); } if (data.candidates && data.candidates[0].content.parts[0].text) { els.aiOutput.innerText = data.candidates[0].content.parts[0].text; } else { els.aiOutput.innerText = 'Błąd: Pusta odpowiedź od modelu.'; } } catch (err) { els.aiOutput.innerText = 'Błąd: ' + err.message; } });
